'wss:' : 'ws:'; const url = `${proto}//${location.host}/ws`; let settled = false; this.ws = new WebSocket(url); const timer = setTimeout(() => { if (!settled) { settled = true; reject(new Error('连接超时，请检查服务器是否启动')); try { this.ws.close(); } catch(e) {} } }, 8000); this.ws.onopen = () => { if (settled) return; settled = true; clearTimeout(timer); this.connected = true; resolve(); }; this.ws.onerror = (e) => { if (settled) return; settled = true; clearTimeout(timer); reject(new Error('连接服务器失败')); }; this.ws.onclose = () => { if (!settled) { // 连接建立前就关闭（比如服务器未启动、端口错误、跨域） settled = true; clearTimeout(timer); reject(new Error('无法连接服务器，请检查服务器是否启动')); return; } this.connected = false; this.emit('disconnected'); }; this.ws.onmessage = (evt) => { try { const msg = JSON.parse(evt.data); this._handleMessage(msg); } catch (e) { console.error('Message parse error:', e); } }; }); } send(type, data) { if (this.ws && this.ws.readyState === 1) { this.ws.send(JSON.stringify({ type, data })); } } _handleMessage(msg) { const { type, data } = msg; switch (type) { case 'roomCreated': this.roomId = data.roomId; this.playerIndex = data.playerIndex; this.isHost = data.isHost; this.emit('roomCreated', data); break; case 'roomJoined': this.roomId = data.roomId; this.playerIndex = data.playerIndex; this.isHost = data.isHost; this.emit('roomJoined', data); break; case 'playerJoined': this.emit('playerJoined', data); break; case 'playerAssigned': this.playerIndex = data.playerIndex; this.isHost = data.isHost; break; case 'gameStart': this.emit('gameStart'); break; case 'opponentDisconnected': this.emit('opponentDisconnected'); break; case 'roomLeft': this.emit('roomLeft'); break; case 'error': this.emit('error', msg.message); break; case 'gameEvent': // 转发游戏事件，保持与 LocalController 一致的接口 this._handleGameEvent(msg.event, data); break; case 'actionRejected': console.warn('Action rejected:', data?.action); break; } } _handleGameEvent(event, data) { // 特殊处理：stateChange 更新本地 state 引用 if (event === 'stateChange') { this.state = data; } this.emit(event, data); } // ---- 房间操作 ---- createRoom() { this.send('createRoom'); } joinRoom(roomId) { this.send('joinRoom', { roomId }); } leaveRoom() { this.send('leaveRoom'); } // ---- 游戏操作（与 LocalController 接口对齐） ---- initGame() { /* 服务端处理 */ } rollDice() { this.send('gameAction', { action: 'rollDice' }); } rerollDice() { this.send('gameAction', { action: 'rerollDice' }); } movePiece(pieceIndex) { this.send('gameAction', { action: 'movePiece', payload: { pieceIndex } }); } selectTask(taskIndex) { this.send('gameAction', { action: 'selectTask', payload: { taskIndex } }); } closeEvent() { this.send('gameAction', { action: 'closeEvent' }); } takeOffEquipment(player, pieceIndex, pos, equipIndex) { this.send('gameAction', { action: 'takeOffEquipment', payload: { pieceIndex, pos, equipIndex } }); } buySkipClothes(player, pieceIndex, pos) { this.send('gameAction', { action: 'buySkipClothes', payload: { pieceIndex, pos } }); } buyItem(itemId) { this.send('gameAction', { action: 'buyItem', payload: { itemId } }); } closeShop() { this.send('gameAction', { action: 'closeShop' }); } // 动画结束回调（本地动画完了通知服务端继续状态机） onDiceAnimationEnd() { // 服务端引擎自己有状态机，骰子动画是纯视觉 } onMoveComplete(player, pieceIndex, fromPos, toPos) { // 服务端引擎自己有状态机，移动动画是纯视觉 } } // ============================================================ // 应用控制器 — 管理页面切换、模式选择、网络连接 // ============================================================ const App = { mode: null, // 'local' | 'online' game: null, ui: null, network: null, init() { this.$ = (id) => document.getElementById(id); this.homeView = this.$('homeView'); this.configView = this.$('configView'); this.gameView = this.$('gameView'); this.lobbyView = this.$('lobbyView'); this.joinModal = this.$('joinModal'); this.disconnectModal = this.$('disconnectModal'); // 首页按钮 this.$('modeLocal').addEventListener('click', () => this.startLocalMode()); this.$('modeCreate').addEventListener('click', () => this.createRoom()); this.$('modeJoin').addEventListener('click', () => this.showJoinModal()); // 加入房间弹窗 this.$('joinModalCancel').addEventListener('click', () => this.hideJoinModal()); this.$('joinModalConfirm').addEventListener('click', () => this.confirmJoinRoom()); this.$('joinRoomInput').addEventListener('keydown', (e) => { if (e.key === 'Enter') this.confirmJoinRoom(); }); // 房间页 this.$('copyRoomId').addEventListener('click', () => this.copyRoomId()); this.$('leaveRoomBtn').addEventListener('click', () => this.leaveRoom()); // 断连弹窗 this.$('disconnectBackBtn').addEventListener('click', () => this.goHome()); this.$('lobbyTitle').textContent = '已加入房间'; this.$('lobbySubtitle').textContent = '等待房主开始游戏'; this.$('lobbyStatus').textContent = '游戏即将开始...'; this.$('lobbyP1Name').textContent = '房主'; this.$('lobbyP2Name').textContent = '你'; this.showView('lobby'); }); net.on('playerJoined', () => { this.$('lobbyP2Name').textContent = '玩家2 已加入'; this.$('lobbyStatus').textContent = '准备开始...'; // 房主自动开始游戏 if (net.isHost) { setTimeout(() => { net.send('startGame');
